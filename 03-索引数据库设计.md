## 一、SQLite 索引数据库 Schema

### 1.1 数据库表结构

```sql
-- 文件索引表
CREATE TABLE files (
    id INTEGER PRIMARY KEY AUTOINCREMENT,
    name TEXT NOT NULL,              -- 文件名
    path TEXT NOT NULL,              -- 完整路径
    directory TEXT NOT NULL,         -- 目录
    size INTEGER DEFAULT 0,          -- 文件大小(字节)
    create_time INTEGER,             -- 创建时间(Unix ms)
    modify_time INTEGER,             -- 修改时间(Unix ms)
    access_time INTEGER,             -- 访问时间(Unix ms)
    attributes INTEGER DEFAULT 0,    -- 文件属性位图
    is_directory INTEGER DEFAULT 0,  -- 是否目录
    drive_id INTEGER,                -- 磁盘ID
    mft_record INTEGER,              -- MFT记录号(NTFS)
    UNIQUE(path) ON CONFLICT REPLACE
);

-- 磁盘表
CREATE TABLE drives (
    id INTEGER PRIMARY KEY AUTOINCREMENT,
    letter TEXT UNIQUE,              -- 盘符 C/D
    label TEXT,                      -- 卷标
    file_system TEXT,                -- NTFS/FAT32/exFAT
    is_ntfs INTEGER DEFAULT 0,       -- 是否NTFS
    enabled INTEGER DEFAULT 1,       -- 是否启用
    last_indexed INTEGER             -- 最后索引时间
);

-- 排除列表
CREATE TABLE exclusions (
    id INTEGER PRIMARY KEY AUTOINCREMENT,
    path TEXT UNIQUE,                -- 排除路径
    type INTEGER DEFAULT 0,          -- 0:文件夹 1:磁盘
    created INTEGER                  -- 创建时间
);

-- 索引元数据
CREATE TABLE metadata (
    key TEXT PRIMARY KEY,
    value TEXT
);

-- 索引
CREATE INDEX idx_files_name ON files(name);
CREATE INDEX idx_files_directory ON files(directory);
CREATE INDEX idx_files_size ON files(size);
CREATE INDEX idx_files_modify_time ON files(modify_time);
```

### 1.2 表结构说明

| 表名 | 说明 |
|------|------|
| `files` | 文件索引表，存储所有文件信息 |
| `drives` | 磁盘表，存储磁盘信息 |
| `exclusions` | 排除列表，存储不索引的目录 |
| `metadata` | 元数据表，存储配置信息 |

## 二、索引数据库接口

### 2.1 接口定义

```cangjie
/// 文件条目
struct FileEntry {
    id: Int64
    name: String
    path: String
    directory: String
    size: Int64
    createTime: Int64
    modifyTime: Int64
    accessTime: Int64
    attributes: Int32
    isDirectory: Bool
    driveId: Int32
}

/// 搜索选项
struct SearchOptions {
    limit: Int32 = 100
    offset: Int32 = 0
    sort: String = "name"
    order: String = "asc"
    caseSensitive: Bool = false
}

/// 搜索结果
struct SearchResult {
    file: FileEntry
    score: Float64
    highlights: Highlight[]
}

/// 索引数据库接口
interface IndexDB {
    open(path: String): Result<Unit, DBError>
    close(): Result<Unit, DBError>
    insertFile(file: FileEntry): Result<Unit, DBError>
    updateFile(file: FileEntry): Result<Unit, DBError>
    deleteFile(path: String): Result<Unit, DBError>
    search(query: String, options: SearchOptions): Result<SearchResult[], DBError>
    getFileCount(): Result<Int64, DBError>
    beginTransaction(): Result<Unit, DBError>
    commit(): Result<Unit, DBError>
}
```

### 2.2 实现类

```cangjie
class IndexDBImpl: IndexDB {
    private var db: Database?
    
    open(path: String): Result<Unit, DBError> {
        db = SQLite.open(path)
        
        // 创建表
        db.execute(CREATE_FILES_TABLE)
        db.execute(CREATE_DRIVES_TABLE)
        db.execute(CREATE_EXCLUSIONS_TABLE)
        db.execute(CREATE_METADATA_TABLE)
        
        // 创建索引
        db.execute(CREATE_IDX_FILES_NAME)
        db.execute(CREATE_IDX_FILES_DIRECTORY)
        
        return Ok(())
    }
    
    insertFile(file: FileEntry): Result<Unit, DBError> {
        let sql = """
            INSERT OR REPLACE INTO files 
            (name, path, directory, size, create_time, modify_time, attributes, is_directory, drive_id)
            VALUES (?, ?, ?, ?, ?, ?, ?, ?, ?)
        """
        
        db.execute(sql, [
            file.name,
            file.path,
            file.directory,
            file.size,
            file.createTime,
            file.modifyTime,
            file.attributes,
            file.isDirectory,
            file.driveId
        ])
        
        return Ok(())
    }
    
    search(query: String, options: SearchOptions): Result<SearchResult[], DBError> {
        let sql = """
            SELECT * FROM files 
            WHERE name LIKE ?
            ORDER BY ${options.sort} ${options.order}
            LIMIT ? OFFSET ?
        """
        
        let pattern = "%" + query + "%"
        let results = db.query(sql, [
            pattern,
            options.limit,
            options.offset
        ])
        
        return Ok(results)
    }
}
```

## 三、增量更新机制

### 3.1 增量更新器

```cangjie
/// 增量更新器
class IncrementalUpdater {
    private var db: IndexDB
    private var pendingChanges: ArrayList<FileChange>()
    
    applyChange(change: FileChangeEvent): Result<Unit, UpdateError> {
        pendingChanges.add(change)
        return Ok(())
    }
    
    batchUpdate(): Result<Unit, UpdateError> {
        db.beginTransaction()
        
        for change in pendingChanges {
            match change.changeType {
                FILE_ADDED => {
                    db.insertFile(change.file)
                }
                FILE_REMOVED => {
                    db.deleteFile(change.path)
                }
                FILE_MODIFIED => {
                    db.updateFile(change.file)
                }
                FILE_RENAMED => {
                    db.updateFile(change.newFile)
                }
            }
        }
        
        db.commit()
        pendingChanges.clear()
        
        return Ok(())
    }
}
```

### 3.2 更新流程

```
┌─────────────────────────────────────────────────────────────────┐
│                    增量更新流程                                 │
└─────────────────────────────────────────────────────────────────┘

1. 监控文件变更
   └─ 使用 ReadDirectoryChangesW API

2. 收集变更事件
   └─ 添加到 pendingChanges 队列

3. 批量更新
   └─ 开始事务
   └─ 应用所有变更
   └─ 提交事务

4. 通知 UI 刷新
   └─ 发送更新完成事件
```

## 四、查询优化

### 4.1 索引策略

| 索引 | 字段 | 用途 |
|------|------|------|
| idx_files_name | name | 文件名搜索 |
| idx_files_directory | directory | 目录搜索 |
| idx_files_size | size | 大小过滤 |
| idx_files_modify_time | modify_time | 时间过滤 |

### 4.2 查询优化

```cangjie
/// 查询优化器
class QueryOptimizer {
    /// 优化查询
    optimizeQuery(query: String): OptimizedQuery {
        // 1. 解析查询
        let plan = parseQuery(query)
        
        // 2. 选择最佳索引
        let index = selectIndex(plan)
        
        // 3. 构建执行计划
        let executionPlan = buildExecutionPlan(plan, index)
        
        return executionPlan
    }
    
    /// 选择索引
    selectIndex(plan: QueryPlan): Index {
        match plan.filterType {
            FILTER_NAME => return idx_files_name
            FILTER_DIRECTORY => return idx_files_directory
            FILTER_SIZE => return idx_files_size
            FILTER_TIME => return idx_files_modify_time
        }
    }
}
```
