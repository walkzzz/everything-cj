## 二、搜索栏设计

### 2.1 搜索栏接口

```cangjie
/// 搜索栏接口
interface SearchBar {
    create(parent: WindowHandle): Result<Unit, UIError>
    getText(): String
    setText(text: String): Unit
    onTextChanged(callback: (String) -> Unit): Unit
    onKeyDown(callback: (Key) -> Unit): Unit
}
```

### 2.2 搜索栏实现

```cangjie
class SearchBarImpl: SearchBar {
    private var handle: ControlHandle?
    private var textChangedCallback: (String) -> Unit?
    private var keyDownCallback: (Key) -> Unit?
    
    create(parent: WindowHandle): Result<Unit, UIError> {
        handle = createEditControl(parent)
        onTextChanged(handle, this::onTextChanged)
        onKeyDown(handle, this::onKeyDown)
        return Ok(())
    }
    
    onTextChanged(text: String): Unit {
        if textChangedCallback != null {
            textChangedCallback(text)
        }
    }
    
    onKeyDown(key: Key): Unit {
        if keyDownCallback != null {
            keyDownCallback(key)
        }
    }
}
```

## 三、结果列表设计

### 3.1 结果列表项

```cangjie
/// 结果列表项
struct ResultItem {
    name: String
    path: String
    size: String
    modifyTime: String
    icon: Icon
}
```

### 3.2 结果列表接口

```cangjie
/// 结果列表接口
interface ResultList {
    create(parent: WindowHandle): Result<Unit, UIError>
    addItem(item: ResultItem): Unit
    clear(): Unit
    getSelectedItem(): ResultItem?
    onItemClick(callback: (ResultItem) -> Unit): Unit
}
```

### 3.3 结果列表实现

```cangjie
class ResultListImpl: ResultList {
    private var handle: ControlHandle?
    private var items: ArrayList<ResultItem>()
    private var itemClickCallback: (ResultItem) -> Unit?
    
    create(parent: WindowHandle): Result<Unit, UIError> {
        handle = createListView(parent)
        return Ok(())
    }
    
    addItem(item: ResultItem): Unit {
        items.add(item)
        addListViewItem(handle, item)
    }
    
    clear(): Unit {
        items.clear()
        clearListView(handle)
    }
}
```

## 四、快捷键设计

### 4.1 快捷键列表

| 快捷键 | 功能 |
|--------|------|
| `Ctrl + Esc` | 显示/隐藏主窗口 |
| `Ctrl + F` | 聚焦搜索栏 |
| `Enter` | 打开选中文件 |
| `Ctrl + O` | 打开所在文件夹 |
| `Ctrl + C` | 复制文件名 |
| `Ctrl + Shift + C` | 复制完整路径 |
| `Esc` | 清除搜索/关闭窗口 |
| `Up/Down` | 上下选择结果 |

### 4.2 快捷键处理

```cangjie
/// 快捷键处理器
class ShortcutHandler {
    handle(key: Key, modifiers: Modifier): Boolean {
        match (key, modifiers) {
            (Key.ESC, Modifier.NONE) => {
                hideWindow()
                return true
            }
            (Key.ENTER, Modifier.NONE) => {
                openSelectedFile()
                return true
            }
            (Key.CTRL, Key.F) => {
                focusSearchBar()
                return true
            }
            _ => return false
        }
    }
}
```
