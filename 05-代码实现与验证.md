### 1.2 MFT 读取器

```cangjie
// core/mft_reader.cj
import stdx.fs.*
import stdx.io.*

struct MFTRecord {
    recordNumber: Int64
    fileName: String
    fullPath: String
    fileSize: Int64
    createTime: DateTime
    modifyTime: DateTime
    attributes: FileAttribute
    isDirectory: Bool
}

class MFTReaderImpl: MFTReader {
    open(drive: Drive): Result<MFTHandle, MFTError> {
        let path = "\\.\" + drive + "\$MFT"
        return File.open(path, OpenMode.READ)
    }
    
    readAll(handle: MFTHandle): Result<MFTRecord[], MFTError> {
        let records = ArrayList<MFTRecord>()
        
        while true {
            let data = handle.read(1024)
            if data.length == 0 {
                break
            }
            
            let record = parseRecord(data)
            if record.fileName != "" {
                records.add(record)
            }
        }
        
        return Ok(records)
    }
}
```

### 1.3 索引引擎

```cangjie
// core/index_engine.cj
import core.mft_reader.*
import storage.index_db.*

class IndexEngineImpl: IndexEngine {
    private var db: IndexDB
    private var reader: MFTReader
    
    constructor(db: IndexDB) {
        this.db = db
        this.reader = MFTReaderImpl()
    }
    
    buildIndex(drive: Drive): Result<Unit, BuildError> {
        let handle = reader.open(drive)
        let records = reader.readAll(handle)
        
        db.beginTransaction()
        
        for record in records {
            let file = FileEntry {
                name: record.fileName,
                path: record.fullPath,
                size: record.fileSize,
                createTime: record.createTime,
                modifyTime: record.modifyTime,
                isDirectory: record.isDirectory
            }
            db.insertFile(file)
        }
        
        db.commit()
        
        return Ok(())
    }
}
```

### 1.4 查询解析器

```cangjie
// search/query_parser.cj
import stdx.regex.*

struct QueryPlan {
    keywords: String[]
    filters: Filter[]
    sortField: String
    sortOrder: String
    limit: Int32
}

class QueryParserImpl: QueryParser {
    parse(input: String): QueryPlan {
        let plan = QueryPlan {
            keywords: ArrayList<String>()
            filters: ArrayList<Filter>()
            sortField: "name"
            sortOrder: "asc"
            limit: 100
        }
        
        let tokens = tokenize(input)
        
        for token in tokens {
            match token {
                FILTER_EXT => {
                    plan.filters.add(Filter.Extension(token.value))
                }
                FILTER_PATH => {
                    plan.filters.add(Filter.Path(token.value))
                }
                KEYWORD => {
                    plan.keywords.add(token.value)
                }
            }
        }
        
        return plan
    }
}
```

### 1.5 索引数据库

```cangjie
// storage/index_db.cj
import sqlite.*

class IndexDBImpl: IndexDB {
    private var db: Database?
    
    open(path: String): Result<Unit, DBError> {
        db = SQLite.open(path)
        db.execute(CREATE_FILES_TABLE)
        db.execute(CREATE_DRIVES_TABLE)
        db.execute(CREATE_EXCLUSIONS_TABLE)
        return Ok(())
    }
    
    insertFile(file: FileEntry): Result<Unit, DBError> {
        let sql = """
            INSERT OR REPLACE INTO files 
            (name, path, directory, size, create_time, modify_time, is_directory, drive_id)
            VALUES (?, ?, ?, ?, ?, ?, ?, ?)
        """
        
        db.execute(sql, [
            file.name, file.path, file.directory,
            file.size, file.createTime, file.modifyTime,
            file.isDirectory, file.driveId
        ])
        
        return Ok(())
    }
    
    search(query: String, options: SearchOptions): Result<SearchResult[], DBError> {
        let sql = """
            SELECT * FROM files 
            WHERE name LIKE ?
            ORDER BY ${options.sort} ${options.order}
            LIMIT ? OFFSET ?
        """
        
        let pattern = "%" + query + "%"
        let results = db.query(sql, [pattern, options.limit, options.offset])
        
        return Ok(results)
    }
}
```

### 1.6 CLI 工具

```cangjie
// cli/es.cj
import stdx.io.*
import search.query_parser.*
import storage.index_db.*

main() {
    let args = getArgs()
    
    if args.length == 0 {
        println("Usage: es.exe [options] <search>")
        return
    }
    
    let query = args[0]
    let parser = QueryParserImpl()
    let plan = parser.parse(query)
    
    let db = IndexDBImpl()
    db.open("everything.db")
    
    let results = db.search(query, SearchOptions {
        limit: 100, sort: "name", order: "asc"
    })
    
    for result in results {
        println("${result.name} (${result.size}) - ${result.path}")
    }
}
```
