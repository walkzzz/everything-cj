## 一、MFT 读取器设计

### 1.1 NTFS MFT 原理

MFT (Master File Table) 是 NTFS 文件系统的核心数据结构：

```
┌─────────────────────────────────────────────────────────────────┐
│                    NTFS MFT 结构解析                             │
└─────────────────────────────────────────────────────────────────┘

MFT 记录 (通常 1KB)
┌─────────────────────────────────────────────────────────────────┐
│  文件记录头 (FILE Header)                                        │
│  - 签名: "FILE"                                                  │
│  - 更新序列号                                                    │
│  - MFT 记录号                                                    │
└─────────────────────────────────────────────────────────────────┘
┌─────────────────────────────────────────────────────────────────┐
│  属性 (Attributes)                                               │
│  ├─ $STANDARD_INFORMATION: 时间戳、权限                          │
│  ├─ $FILE_NAME: 文件名、父目录引用                               │
│  ├─ $DATA: 文件数据 (小文件直接存储)                             │
│  └─ $ATTRIBUTE_LIST: 属性列表 (大文件)                           │
└─────────────────────────────────────────────────────────────────┘
```

### 1.2 MFT 读取器接口

```cangjie
/// MFT 记录结构
struct MFTRecord {
    recordNumber: Int64       // 记录号
    fileName: String          // 文件名
    fullPath: String          // 完整路径
    fileSize: Int64           // 文件大小
    createTime: DateTime      // 创建时间
    modifyTime: DateTime      // 修改时间
    attributes: FileAttribute // 文件属性
    isDirectory: Bool         // 是否目录
}

/// MFT 读取器接口
interface MFTReader {
    /// 打开 MFT 文件
    open(drive: Drive): Result<MFTHandle, MFTError>
    
    /// 读取所有 MFT 记录
    readAll(handle: MFTHandle): Result<MFTRecord[], MFTError>
    
    /// 增量读取（从指定偏移）
    readDelta(handle: MFTHandle, offset: Int64): Result<MFTRecord[], MFTError>
    
    /// 解析 MFT 记录
    parseRecord(data: Byte[]): MFTRecord
}
```

```cangjie
class MFTReaderImpl: MFTReader {
    open(drive: Drive): Result<MFTHandle, MFTError> {
        let path = "\\.\" + drive + "\$MFT"
        return File.open(path, OpenMode.READ)
    }
    
    readAll(handle: MFTHandle): Result<MFTRecord[], MFTError> {
        let records = ArrayList<MFTRecord>()
        let offset = 0
        
        while true {
            let data = handle.read(1024)
            if data.length == 0 {
                break
            }
            
            let record = parseRecord(data)
            if record.fileName != "" {
```

## 二、文件系统监控设计

### 2.1 文件变更类型

```cangjie
/// 文件变更类型
enum FileChange {
    FILE_ADDED
    FILE_REMOVED
    FILE_MODIFIED
    FILE_RENAMED
}

/// 文件变更事件
struct FileChangeEvent {
    changeType: FileChange
    path: String
    oldPath: String?       // 重命名时使用
    timestamp: DateTime
}
```

### 2.2 文件监控接口

```cangjie
/// 文件监控接口
interface FileMonitor {
    /// 启动监控
    start(path: String): Result<Unit, MonitorError>
    
    /// 停止监控
    stop(): Result<Unit, MonitorError>
    
    /// 注册变更回调
    onChange(callback: (FileChangeEvent) -> Unit): Unit
}
```

### 2.3 文件监控实现

```cangjie
class FileMonitorImpl: FileMonitor {
    private var running: Bool = false
    private var callbacks: ArrayList<(FileChangeEvent) -> Unit>()
    private var thread: Thread?
    
    start(path: String): Result<Unit, MonitorError> {
        if running {
            return Err(MonitorError.ALREADY_RUNNING)
        }
        
        running = true
        thread = Thread.new(() => {
            monitorLoop(path)
        })
        
        return Ok(())
    }
    
    stop(): Result<Unit, MonitorError> {
        running = false
        if thread != null {
            thread.join()
        }
        return Ok(())
    }
    
    private monitorLoop(path: String): Unit {
        while running {
            let changes = readDirectoryChanges(path)
            for change in changes {
                let event = FileChangeEvent {
                    changeType: change.type,
                    path: change.path,
                    timestamp: DateTime.now()
                }
                for callback in callbacks {
                    callback(event)
                }
            }
        }
    }
}
```

## 三、HTTP API 服务器设计

### 3.1 HTTP 服务器接口

```cangjie
/// HTTP 服务器接口
interface HTTPServer {
    start(port: Int16): Result<Unit, ServerError>
    stop(): Result<Unit, ServerError>
    registerRoute(method: String, path: String, handler: (HTTPRequest) -> HTTPResponse): Unit
}
```

### 3.2 HTTP 服务器实现

```cangjie
class HTTPServerImpl: HTTPServer {
    private var server: HTTPServer?
    private var routes: HashMap<String, (HTTPRequest) -> HTTPResponse>()
    
    start(port: Int16): Result<Unit, ServerError> {
        server = HTTPServerBuilder()
            .port(port)
            .handler(this::handleRequest)
            .build()
        
        server.start()
        return Ok(())
    }
    
    private handleRequest(request: HTTPRequest): HTTPResponse {
        let key = request.method + ":" + request.path
        if routes.containsKey(key) {
            return routes[key](request)
        }
        return HTTPResponse { status: 404, body: "Not Found" }
    }
}
```
